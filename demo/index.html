binary-search.strukt
title: Binary Search

# Returns the index of target in a sorted array, or -1 if not found

set low to 0
set high to length of array - 1
set result to -1

while low <= high:
    set mid to floor((low + high) / 2)
    if array[mid] == target:
        set result to mid
        break
    else if array[mid] < target:
        set low to mid + 1
    else:
        set high to mid - 1

return result
bubble-sort.strukt
title: Bubble Sort

# Sorts array in-place in ascending order

set n to length of array

for i from 0 to n - 1:
    for j from 0 to n - i - 2:
        if array[j] > array[j + 1]:
            swap array[j] and array[j + 1]

return array

title: Parallel File Download and Verify

parallel:
    thread:
        # Download and verify file A
        call download("file-a.zip", destinationA)
        call verify(destinationA, checksumA)
        if verification failed:
            call flagError("file-a checksum mismatch")
    thread:
        # Download and verify file B
        call download("file-b.zip", destinationB)
        call verify(destinationB, checksumB)
        if verification failed:
            call flagError("file-b checksum mismatch")
    thread:
        # Download and verify file C
        call download("file-c.zip", destinationC)
        call verify(destinationC, checksumC)
        if verification failed:
            call flagError("file-c checksum mismatch")

call reportResults()
